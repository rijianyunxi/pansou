# 用法（在仓库根目录）：
#   cp links.example.txt links.txt
#   npm run regress links.txt                 # 依次转存并统计耗时

#   npm run regress links.txt -- --no-top-fids  # A/B 对照：强制走「列目录找回 fid」
#   npm run regress links.txt -- --json out.json
#
